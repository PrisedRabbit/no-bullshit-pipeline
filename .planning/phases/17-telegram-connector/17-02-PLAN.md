**8. Telegram wizard state machine** — add after the Linear wizard code block (around line 1200+):

```js
// ── Telegram Wizard ──────────────────────────────────────────────────────────

let telegramWizardState = {
  step: 0,
  integrationId: null,
  error: null,
  chats: [],           // Vec<TelegramChat> from resolve_telegram_chats
  selectedChatId: null,
  selectedChatTitle: '',
  selectedChatType: '',
};

function resetTelegramWizardState() {
  telegramWizardState = {
    step: 0, integrationId: null, error: null,
    chats: [], selectedChatId: null, selectedChatTitle: '', selectedChatType: '',
  };
}

function closeTelegramWizard() {
  resetTelegramWizardState();
  const modal = document.getElementById('telegram-wizard-modal');
  if (modal) modal.style.display = 'none';
}

function openTelegramWizard() {
  resetTelegramWizardState();
  const modal = document.getElementById('telegram-wizard-modal');
  if (!modal) return;
  modal.style.display = 'flex';

  const cancelBtn = document.getElementById('telegram-wizard-cancel');
  const cancelClone = cancelBtn.cloneNode(true);
  cancelBtn.parentNode.replaceChild(cancelClone, cancelBtn);
  cancelClone.addEventListener('click', async () => {
    if (telegramWizardState.integrationId) {
      try {
        await window.__TAURI__.core.invoke('remove_telegram_integration', {
          integrationId: telegramWizardState.integrationId,
        });
      } catch (_) {}
    }
    closeTelegramWizard();
  });

  renderTelegramWizardStep();
}

function replaceTelegramNextBtn(handler) {
  const nextBtn = document.getElementById('telegram-wizard-next');
  const clone = nextBtn.cloneNode(true);
  nextBtn.parentNode.replaceChild(clone, nextBtn);
  clone.addEventListener('click', handler);
  return clone;
}

function renderTelegramWizardStep() {
  const body = document.getElementById('telegram-wizard-body');
  const progress = document.getElementById('telegram-wizard-progress');
  const title = document.getElementById('telegram-wizard-title');

  const step = telegramWizardState.step;
  const progressPct = step === 0 ? 50 : 100;
  progress.style.width = progressPct + '%';

  const errorHtml = telegramWizardState.error
    ? `<div class="wizard-error" style="color:var(--danger);margin-bottom:12px;">${escapeHtml(telegramWizardState.error)}</div>`
    : '';

  if (step === 0) {
    // Step 0: Bot token entry
    title.textContent = 'Step 1 of 2 — Bot Token';
    body.innerHTML = `
      ${errorHtml}
      <div class="input-group">
        <label>Integration Name</label>
        <input id="tg-name-input" type="text" placeholder="e.g. My Telegram Bot" value="" />
      </div>
      <div class="input-group" style="margin-top:12px;">
        <label>Bot Token</label>
        <input id="tg-token-input" type="password" placeholder="123456:ABC-DEF..." value="" />
        <div class="input-hint" style="margin-top:4px;font-size:0.8em;color:var(--text-secondary);">
          Get a token from <a href="#" onclick="return false;">@BotFather</a> on Telegram
        </div>
      </div>`;

    replaceTelegramNextBtn(async () => {
      const name = document.getElementById('tg-name-input').value.trim();
      const token = document.getElementById('tg-token-input').value.trim();
      if (!name) { telegramWizardState.error = 'Please enter a name for this integration.'; renderTelegramWizardStep(); return; }
      if (!token) { telegramWizardState.error = 'Please enter a bot token.'; renderTelegramWizardStep(); return; }
      telegramWizardState.error = null;
      try {
        const id = await window.__TAURI__.core.invoke('add_telegram_integration', { name, botToken: token });
        telegramWizardState.integrationId = id;
        // Immediately resolve chats so step 1 can render them
        try {
          telegramWizardState.chats = await window.__TAURI__.core.invoke('resolve_telegram_chats', { integrationId: id });
        } catch (_) {
          telegramWizardState.chats = [];
        }
        telegramWizardState.step = 1;
        renderTelegramWizardStep();
      } catch (err) {
        telegramWizardState.error = String(err);
        renderTelegramWizardStep();
      }
    });

  } else if (step === 1) {
    // Step 1: Chat selection (from getUpdates or manual entry)
    title.textContent = 'Step 2 of 2 — Select Chat';
    const chats = telegramWizardState.chats;

    let chatListHtml = '';
    if (chats.length > 0) {
      chatListHtml = `<div style="margin-bottom:12px;">
        <label style="display:block;margin-bottom:8px;">Chats where your bot has received messages:</label>
        <div id="tg-chat-list">`;
      for (const chat of chats) {
        chatListHtml += `
          <div class="tg-chat-option" data-chat-id="${chat.id}" data-chat-title="${escapeHtml(chat.title)}" data-chat-type="${escapeHtml(chat.chat_type)}"
               style="padding:8px 12px;border:1px solid var(--border);border-radius:6px;margin-bottom:6px;cursor:pointer;">
            <strong>${escapeHtml(chat.title)}</strong>
            <span style="margin-left:8px;font-size:0.8em;color:var(--text-secondary);">${escapeHtml(chat.chat_type)} · ${chat.id}</span>
          </div>`;
      }
      chatListHtml += `</div></div>
        <div style="font-size:0.85em;color:var(--text-secondary);margin-bottom:8px;">Or enter a chat ID manually:</div>`;
    } else {
      chatListHtml = `<div class="info-box" style="margin-bottom:12px;">
        No chats found automatically. Send a message to your bot first, or enter a chat ID manually below.
        </div>`;
    }

    body.innerHTML = `
      ${errorHtml}
      ${chatListHtml}
      <div class="input-group">
        <label>Chat ID (manual)</label>
        <input id="tg-chatid-input" type="text" placeholder="-100123456789 or 123456789" value="${telegramWizardState.selectedChatId || ''}" />
      </div>
      <div class="input-group" style="margin-top:8px;">
        <label>Chat Label</label>
        <input id="tg-chattitle-input" type="text" placeholder="e.g. My Group Chat" value="${escapeHtml(telegramWizardState.selectedChatTitle)}" />
      </div>`;

    // Wire chat option click handlers
    body.querySelectorAll('.tg-chat-option').forEach(el => {
      el.addEventListener('click', () => {
        body.querySelectorAll('.tg-chat-option').forEach(o => o.style.background = '');
        el.style.background = 'var(--accent-muted, rgba(99,102,241,0.1))';
        document.getElementById('tg-chatid-input').value = el.dataset.chatId;
        document.getElementById('tg-chattitle-input').value = el.dataset.chatTitle;
        telegramWizardState.selectedChatId = parseInt(el.dataset.chatId, 10);
        telegramWizardState.selectedChatTitle = el.dataset.chatTitle;
        telegramWizardState.selectedChatType = el.dataset.chatType;
      });
    });

    const nextBtn = replaceTelegramNextBtn(async () => {
      const rawId = document.getElementById('tg-chatid-input').value.trim();
      const label = document.getElementById('tg-chattitle-input').value.trim();
      if (!rawId) { telegramWizardState.error = 'Please select or enter a chat ID.'; renderTelegramWizardStep(); return; }
      const chatId = parseInt(rawId, 10);
      if (isNaN(chatId)) { telegramWizardState.error = 'Chat ID must be a number.'; renderTelegramWizardStep(); return; }
      if (!label) { telegramWizardState.error = 'Please enter a label for this chat.'; renderTelegramWizardStep(); return; }
      telegramWizardState.error = null;
      try {
        await window.__TAURI__.core.invoke('save_telegram_chat', {
          integrationId: telegramWizardState.integrationId,
          chatId,
          chatTitle: label,
          chatType: telegramWizardState.selectedChatType || 'unknown',
        });
        await loadAllIntegrations();
        renderConnectedIntegrations();
        closeTelegramWizard();
      } catch (err) {
        telegramWizardState.error = String(err);
        renderTelegramWizardStep();
      }
    });
    nextBtn.textContent = 'Finish';
  }
}
```
